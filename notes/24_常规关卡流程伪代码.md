## 2. 主循环（每帧 tick，FixedUpdate 1/30s）

```python
def battle_tick():
    # —— 阶段 1：费用与待部署区 ——
    if frame % 30 == 0:                       # 每秒
        cost = min(cost + COST_RECOVER * cost_multiplier, COST_MAX)  # [00 总览] 倍率可被关卡条件/合约改

    # —— 阶段 2：波次出生 ——
    for action in due_actions():              # 波次时间轴到点
        spawn_enemy(action)                    # 生成敌人（出生偏移→实体坐标/光标坐标）

    # —— 阶段 3：实体 tick（敌人→干员→装置，顺序影响同帧判定）——
    for enemy in enemies:     enemy_tick(enemy)
    for operator in operators: operator_tick(operator)
    for device in devices:    device_tick(device)

    # —— 阶段 4：buff 统一收尾（延迟结束）[20 cv17305212]——
    flush_buff_end_events()                   # 标记结束的 buff 在此统一移除/回收

    # —— 阶段 5：结算检查 ——
    if target_hp <= 0:        return FAIL
    if all_waves_cleared and all_enemies_dead: return WIN
    frame += 1
```

## 3. 敌人 tick（行动核心）

```python
def enemy_tick(e):
    # 3.1 状态机优先：攻击/被阻挡/失衡/眩晕/睡眠/死亡等非 MOVE 状态不自主移动 [20 cv31743470]
    if e.state != MOVE: return

    # 3.2 寻路移动（每帧）：
    #   0. 实体坐标在地图外 → 直接以理论移速回地图
    #   1. 根据光标坐标查 nextnode 确定目标/给定方向 [21 cv44041657]
    target = next_target(e)                   # 非移动检查点→方向为 0；移动类→光标→nextnode 单位向量
    #   1.5 一帧内可抵达目标 → 直接设置到目标
    #   2.5 束缚检查：束缚则本帧位移=0 [02 §7.5 束缚]
    #   3. 避障力（每 3 帧计算一次并缓存）[20 cv31743470]
    avoid = avoid_force(e) if frame % 3 == 0 else e.cached_avoid
    #   4. 惯性 + 加速度：[20 cv31743470] 地面 steeringFactor=8 / 飞行=20，maxSteeringForce=10/100
    accel = clamp( (desired_dir * v_theory - e.inertia) * steering + avoid, max_steering )
    e.velocity = clamp(accel * FRAME + e.inertia, v_theory)   # v_theory = 属性移速 × 0.5（moveMultiplier）
    e.inertia = e.velocity
    e.pos += e.velocity * FRAME
    #   5. 修正：跨格进入不可通行地块 → 位移向量反射弹出（正面弹回/斜角可擦入）

    # 3.3 检查点/终点判定（每帧，移动后）[20 cv31743470]
    if e.cursor_in(target_checkpoint, r=0.05):      # 移动检查点完成半径 0.05（visitEveryNodeStably 为 0.25）
        advance_checkpoint(e)
    if e.cursor_in(route_end, r=0.05) or e.completed_all_checkpoints:
        enter_target_point(e)                        # 扣除目标生命 1，敌人消失 [02 §3.2]
        target_hp -= 1

    # 3.4 阻挡判定 [99 标注：🔴 需 IDA；来源 PRTS + B站 cv25694535]
    if not e.is_blocked:
        for op in nearby_operators(e, r=op.block_radius):   # 阻挡半径 0.7071，每 3 帧扫描，优先最近
            if can_block(op, e):                     # 阻挡数未满、e 非"无法被阻挡"
                block(e, op); break
    # 3.5 攻击（若被阻挡或进入攻击范围）
    if e.attack_cd <= 0 and target_in_range(e):
        start_attack(e)                              # 前摇→伤害→后摇；间隔帧化向上取整（🟡 K-004）

    # 3.6 技能（CD/优先级；可被沉默）[02 §2.5、§7.12]
    if e.skill_ready and not e.silenced:
        trigger_enemy_skill(e)
```

### 例 A：敌人从出生到进点的完整事件链
```
Action 到点 → spawn_enemy
  → 生成：实体坐标=出生点+偏移；光标坐标=地块中心 [20]
  → 状态机 MOVE，每帧寻路（给定方向+避障+惯性）
  → 途中被干员阻挡：状态→被阻挡，每 3 帧阻挡半径判定成立
     → 触发 buff 事件 ON_OWNER_BLOCKEE_CHANGED [16 解包 隐身敌人案例]
     → 攻击：普攻前摇（帧化）→ 弹道/近战命中 → 伤害结算（见 §5）→ 后摇 → 冷却
  → 阻挡解除（干员撤退/击杀）→ 恢复 MOVE（惯性保留，可能"刹不住入坑" [20 cv31743470]）
  → 光标进终点 0.05 半径 → enter_target_point → target_hp -= 1 → 敌人移除 → 触发"进点"事件
```

```python
def operator_tick(op):
    # 4.1 技能 SP 回复（自动回复/攻击回复/受击回复；阻回=阻挡时回复）[02 §7.12]
    op.sp += sp_per_second(op) * FRAME      # 或攻击/受击时 +n
    if op.sp >= op.skill_cost and op.skill_auto_trigger:
        activate_skill(op)

    # 4.2 索敌 [02 §7.1、21 cv39950636 仇恨]
    targets = acquire_targets(op)           # 按仇恨过滤器：优先级链→阻挡最高→仇恨值（0.1 精度 3 帧一组打先创建）→最早出现
    # 4.3 攻击
    if op.attack_cd <= 0 and targets:
        attack(op, pick_target(targets))    # 攻击流程见 §5；攻击间隔=理论间隔帧化（01 §4）
    op.attack_cd -= FRAME

    # 4.4 天赋（光环类被动每帧/事件驱动）[20 cv7020907 光环]
    for talent in op.aura_talents:
        update_aura(talent)                 # 碰撞进出 + m_invalidTarget 10 帧更新
```

## 5. 伤害结算（一次攻击/技能命中）

```python
def apply_damage(source, target, atk, atk_scale, dmg_type, dmg_scale, ignore_def=0, ignore_def_pct=0):
    # 1. 生成 modifier（攻击属性数据包）[02 §2.1]
    m = Modifier(atk=atk, atk_scale=atk_scale, dmg_type=dmg_type)

    # 2. 来源/目标事件链（顺序敏感，优先级见 01 §12.7 伤判）[01 §12.7]
    emit(source, ON_CALCULATE_DAMAGE, m)        # 攻击倍率调整
    emit(target, ON_BEFORE_APPLYING_MODIFIER, m)
    emit(source, ON_OUTPUT_MODIFIER, m)         # 治疗倍率调整（若治疗）
    emit(source, ON_OUTPUT_DAMAGE, m)           # 伤害倍率调整（脆弱/易伤类）
    emit(target, ON_APPLYING_MODIFIER, m)

    # 3. 伤判效果（事件优先级 > 获取顺序）[01 §12.7]
    if roll(target.evade, dmg_type):      return MISS        # 闪避（多闪避叠乘 [01 §8]）
    if target.block_damage:               return BLOCK       # 抵挡
    if target.barrier > 0:                                 # 屏障（先加易伤再屏障/反之顺序影响 [02 §2.1]）
        absorb = min(m.damage, target.barrier); target.barrier -= absorb; m.damage -= absorb

    # 4. 抗性结算 + 保底 [01 §3]
    if dmg_type == PHYSICAL:
        m.damage = max(0.05 * m.atk, m.atk * atk_scale - (1-ignore_pct)*max(0, target.def - ignore_def))
    elif dmg_type == MAGICAL:
        m.damage = max(0.05 * m.atk, 0.01 * m.atk * atk_scale * max(0, 100 - (1-ignore_pct)*max(0, target.res - ignore_def)))
        # 注：0.01 实为 0.00999999978（定点数）→ 伤害略低 [01 §12.8、20 cv14462417]
    elif dmg_type == PURE:      m.damage = m.atk * atk_scale
    elif dmg_type == ELEMENTAL: m.damage = max(0.05*m.atk, 0.01*m.atk*atk_scale*max(0, 100 - target.ep_damage_res))
                                # 脆弱等对元素伤害不生效 [20 cv24879653，🟡 K-012]

    # 5. 减伤（最终效果-伤害乘算，百分比相加）[02 §2.6]
    m.damage *= (1 - target.dmg_reduction)

    # 6. 应用 + 后续事件
    target.hp -= m.damage
    emit(target, ON_TAKE_DAMAGE, m)             # 受击回复/反伤等
    emit(source, ON_AFTER_OUTPUT_DAMAGE, m)     # 造成伤害触发（击杀回费/回SP等）
    if target.hp <= 0:
        kill(target, source)                    # 死亡判定（注意定点数边界"剩0血不死"容差 [01 §12.8]）
        emit(target, ON_DEATH, source)          # 击杀事件：掉落/计数/回费/触发下一波
```

## 7. 事件触发/结束的通用模型（知识库归纳）

```
触发源：帧 tick / 攻击命中 / 阻挡状态变化 / 部署/撤退 / buff 结束 / 波次节点 / 敌人死亡
  → 事件（如 ON_TAKE_DAMAGE / ON_OWNER_BLOCKEE_CHANGED / ON_BUFF_FINISH）
  → 监听者（天赋/技能/装置/地图机制）按注册执行
  → buff 生命周期：创建（对象池 allocate）→ 生效（可被标记结束但延迟移除）→ 事件末尾统一回收（入栈 recycle）[20]
结束路径：持续到期 / 被移除（净化解控）/ 来源退场 / 对象死亡 / 关卡结束 —— 均经"标记→事件末尾清理"两阶段
```
